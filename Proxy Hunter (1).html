Setup checker (Cloudflare Worker example) 
import { connect } from 'cloudflare:sockets';
export default { async fetch(req) {
  const u = new URL(req.url), p = u.searchParams.get('proxy'), t = u.searchParams.get('type') || 'http';
  const out = o => new Response(JSON.stringify(o), { headers: { 'content-type': 'application/json', 'access-control-allow-origin': '*' } });
  if (!p) return out({ alive: false });
  const [host, port] = p.split(':'), t0 = Date.now();
  try {
    const s = connect({ hostname: host, port: +port }), w = s.writable.getWriter(), r = s.readable.getReader();
    const to = new Promise((_, j) => setTimeout(() => j('timeout'), +u.searchParams.get('timeout') || 8000));
    const run = (async () => {
      if (t === 'socks5') await w.write(new Uint8Array([5, 1, 0]));
      else if (t === 'socks4') await w.write(new Uint8Array([4, 1, 0, 80, 93, 184, 216, 34, 0]));
      else await w.write(new TextEncoder().encode('GET http://example.com/ HTTP/1.1\r\nHost: example.com\r\nConnection: close\r\n\r\n'));
      const { value } = await r.read();
      return t === 'socks5' ? value[0] === 5 : t === 'socks4' ? value[1] === 90 : new TextDecoder().decode(value).startsWith('HTTP/');
    })();
    const alive = await Promise.race([run, to]);
    try { s.close(); } catch (e) {}
    return out({ alive, latency: Date.now() - t0, country: req.cf?.country || '' });
  } catch (e) { return out({ alive: false }); }
} };
